import { PLANS, FREE_SIGNUP_CREDITS, type PlanId } from "@/lib/plans";

export const EMAIL_IMAGE_BASE = "https://assets.motio2edit.com/email/" as const;
export type EmailResult = { subject: string; html: string };

export function fillTemplate(raw: string, vars: Record<string, string | number>): string {
  return raw.replace(/\{\{\s*([a-zA-Z0-9_]+)\s*\}\}/g, (_, k) =>
    vars[k] !== undefined && vars[k] !== null ? String(vars[k]) : `{{${k}}}`,
  );
}

function formatUsd(n: number): string {
  if (n === 0) return "Free";
  if (Number.isInteger(n)) return `$${n}`;
  return `$${n.toFixed(2)}`;
}

function formatCredits(n: number): string {
  return n.toLocaleString("en-US");
}

function planById(id: PlanId) {
  return PLANS.find((p) => p.id === id)!;
}

/** Plan prices + credits sourced from @/lib/plans (single source of truth). */
export function buildPlanEmailVars(): Record<string, string> {
  const free = planById("free");
  const lite = planById("lite");
  const plus = planById("plus");
  const pro = planById("pro");
  const studio = planById("studio");
  const business = planById("business");
  return {
    price_free: formatUsd(free.price.USD),
    price_lite: formatUsd(lite.price.USD),
    price_plus: formatUsd(plus.price.USD),
    price_pro: formatUsd(pro.price.USD),
    price_studio: formatUsd(studio.price.USD),
    price_business: formatUsd(business.price.USD),
    credits_free: formatCredits(free.credits),
    credits_lite: formatCredits(lite.credits),
    credits_plus: formatCredits(plus.credits),
    credits_pro: formatCredits(pro.credits),
    credits_studio: formatCredits(studio.credits),
    credits_business: formatCredits(business.credits),
    free_credits: formatCredits(FREE_SIGNUP_CREDITS),
    plans_url: "https://motio2edit.com/pricing",
  };
}

export const DEFAULT_PLAN_EMAIL_VARS: Record<string, string> = buildPlanEmailVars();
